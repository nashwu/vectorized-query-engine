cmake_minimum_required(VERSION 3.20)
project(vqe VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(VQE_PARQUET "Build Arrow/Parquet integration (requires local packages)" OFF)
add_library(vqe src/vector.cpp src/expression.cpp src/kernels.cpp src/kernels_scalar.cpp src/operators.cpp src/hash.cpp src/blocking.cpp src/statistics.cpp src/plan.cpp src/workloads.cpp)
target_include_directories(vqe PUBLIC include)
target_compile_options(vqe PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(VQE_PARQUET)
  include(cmake/ArrowLocal.cmake)
  target_sources(vqe PRIVATE src/parquet.cpp)
  target_compile_definitions(vqe PUBLIC VQE_HAS_PARQUET=1)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  set_source_files_properties(src/kernels_scalar.cpp PROPERTIES COMPILE_OPTIONS "-fno-vectorize;-fno-slp-vectorize")
elseif(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  set_source_files_properties(src/kernels_scalar.cpp PROPERTIES COMPILE_OPTIONS "-fno-tree-vectorize")
endif()
include(CTest)
add_executable(vqe_demo examples/demo.cpp)
target_link_libraries(vqe_demo PRIVATE vqe)
if(BUILD_TESTING)
  add_executable(vqe_tests tests/test_main.cpp)
  target_link_libraries(vqe_tests PRIVATE vqe)
  add_test(NAME vqe_tests COMMAND vqe_tests)
endif()
